===== 3. Складені (compound) індекси — правило ESR на books_perf =====

// без індексів
{
  plan: 'SORT -> COLLSCAN',
  nReturned: 20,
  totalKeysExamined: 0,
  totalDocsExamined: 200000,
  executionTimeMillis: 138
}

// одиночний {genres:1}: сортування в пам'яті
{
  plan: 'SORT -> FETCH -> IXSCAN(genres_1)',
  nReturned: 20,
  totalKeysExamined: 40000,
  totalDocsExamined: 40000,
  executionTimeMillis: 72
}

// compound {genres:1, available:1, rating:-1}
{
  plan: 'LIMIT -> FETCH -> IXSCAN(genres_1_available_1_rating_-1)',
  nReturned: 20,
  totalKeysExamined: 20,
  totalDocsExamined: 20,
  executionTimeMillis: 8
}
